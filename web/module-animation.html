3. CMake Integration & Compilation

To integrate LabAnimation.dll into your game client, level editor, or standalone tool, link against its import library target in CMake:

# Link against LabAnimation.dll in CMakeLists.txt
find_package(LabEngine REQUIRED)

target_include_directories(MyTarget PRIVATE
    "${LAB_ENGINE_DIR}/include"
)

target_link_libraries(MyTarget PRIVATE
    LabAnimation
)

# Automatically copy LabAnimation.dll to the output directory on build
add_custom_command(TARGET MyTarget POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${LAB_BIN_DIR}/LabAnimation.dll"
    "$<TARGET_FILE_DIR:MyTarget>/LabAnimation.dll"
)
